local function folderExists(path)
	if type(isfolder) == "function" then
		local ok, v = pcall(isfolder, path)
		if ok then return v == true end
	end
	return false
end

local function ensureFolder(path)
	if folderExists(path) then return true end
	if type(makefolder) ~= "function" then return false end
	local ok = pcall(makefolder, path)
	if ok and folderExists(path) then return true end
	return folderExists(path)
end

local function sanitizeNumber(n)
	n = tonumber(n)
	if not n then return 0 end
	if n ~= n then return 0 end
	if n == math.huge or n == -math.huge then return 0 end
	return n
end

local function sanitizeForJSON(v, depth)
	depth = depth or 0
	if depth > 40 then return nil end
	local t = type(v)
	if t == "nil" then return nil end
	if t == "boolean" then return v end
	if t == "string" then return v end
	if t == "number" then return sanitizeNumber(v) end
	if t == "table" then
		local out = {}
		for k, val in pairs(v) do
			local sk
			if type(k) == "string" or type(k) == "number" then
				sk = k
			else
				sk = tostring(k)
			end
			out[sk] = sanitizeForJSON(val, depth + 1)
		end
		return out
	end
	return tostring(v)
end

local function encode(v)
	local cleaned = sanitizeForJSON(v)
	local ok, result = pcall(function()
		return HttpService:JSONEncode(cleaned)
	end)
	if ok and type(result) == "string" then return result end
	return nil
end

local function decode(v)
	if type(v) ~= "string" or v == "" then return nil end
	local ok, result = pcall(function()
		return HttpService:JSONDecode(v)
	end)
	if ok then return result end
	return nil
end

local function readJSON(path)
	if type(readfile) ~= "function" then return nil end
	local ok, data = pcall(readfile, path)
	if not ok or type(data) ~= "string" then return nil end
	return decode(data)
end

local function writeRaw(path, data)
	if type(writefile) ~= "function" then return false end
	local ok = pcall(writefile, path, data)
	return ok
end

local function writeJSON(path, data)
	local encoded = encode(data)
	if not encoded then return false, "JSON encode failed" end
	if writeRaw(path, encoded) then
		return true, ""
	end
	task.wait(0.03)
	if writeRaw(path, encoded) then
		return true, ""
	end
	return false, "writefile failed after retries"
end

local function deleteFile(path)
	if type(delfile) == "function" then
		pcall(delfile, path)
	end
end

local function listFolder(path)
	if type(listfiles) ~= "function" then return {} end
	local ok, result = pcall(listfiles, path)
	if ok and type(result) == "table" then return result end
	return {}
end

local function basename(path)
	local s = tostring(path):gsub("\\", "/")
	return s:match("([^/]+)$") or s
end

local function joinPath(folder, file)
	folder = tostring(folder or ""):gsub("\\", "/"):gsub("/+$", "")
	file = tostring(file or ""):gsub("\\", "/"):gsub("^/+", "")
	if folder == "" then return file end
	if file == "" then return folder end
	return folder .. "/" .. file
end

local function serialize(v)
	local t = typeof(v)
	if v == nil then return { __type = "nil" } end
	if t == "string" then return v end
	if t == "boolean" then return v end
	if t == "number" then return sanitizeNumber(v) end
	if t == "Vector2" then return { __type = "Vector2", X = sanitizeNumber(v.X), Y = sanitizeNumber(v.Y) } end
	if t == "Vector3" then return { __type = "Vector3", X = sanitizeNumber(v.X), Y = sanitizeNumber(v.Y), Z = sanitizeNumber(v.Z) } end
	if t == "Vector2int16" then return { __type = "Vector2int16", X = sanitizeNumber(v.X), Y = sanitizeNumber(v.Y) } end
	if t == "Vector3int16" then return { __type = "Vector3int16", X = sanitizeNumber(v.X), Y = sanitizeNumber(v.Y), Z = sanitizeNumber(v.Z) } end
	if t == "Color3" then return { __type = "Color3", R = sanitizeNumber(v.R), G = sanitizeNumber(v.G), B = sanitizeNumber(v.B) } end
	if t == "UDim" then return { __type = "UDim", Scale = sanitizeNumber(v.Scale), Offset = sanitizeNumber(v.Offset) } end
	if t == "UDim2" then return { __type = "UDim2", XScale = sanitizeNumber(v.X.Scale), XOffset = sanitizeNumber(v.X.Offset), YScale = sanitizeNumber(v.Y.Scale), YOffset = sanitizeNumber(v.Y.Offset) } end
	if t == "BrickColor" then return { __type = "BrickColor", Number = sanitizeNumber(v.Number) } end
	if t == "CFrame" then
		local c = { v:GetComponents() }
		for i = 1, #c do
			c[i] = sanitizeNumber(c[i])
		end
		return { __type = "CFrame", Components = c }
	end
	if t == "NumberRange" then return { __type = "NumberRange", Min = sanitizeNumber(v.Min), Max = sanitizeNumber(v.Max) } end
	if t == "NumberSequence" then
		local points = {}
		for _, p in ipairs(v.Keypoints) do
			points[#points + 1] = {
				Time = sanitizeNumber(p.Time),
				Value = sanitizeNumber(p.Value),
				Envelope = sanitizeNumber(p.Envelope)
			}
		end
		return { __type = "NumberSequence", Keypoints = points }
	end
	if t == "ColorSequence" then
		local points = {}
		for _, p in ipairs(v.Keypoints) do
			points[#points + 1] = { Time = sanitizeNumber(p.Time), Value = serialize(p.Value) }
		end
		return { __type = "ColorSequence", Keypoints = points }
	end
	if t == "PhysicalProperties" then
		return {
			__type = "PhysicalProperties",
			Density = sanitizeNumber(v.Density),
			Friction = sanitizeNumber(v.Friction),
			Elasticity = sanitizeNumber(v.Elasticity),
			FrictionWeight = sanitizeNumber(v.FrictionWeight),
			ElasticityWeight = sanitizeNumber(v.ElasticityWeight)
		}
	end
	if t == "EnumItem" then
		return { __type = "EnumItem", EnumType = tostring(v.EnumType), Name = v.Name, Value = sanitizeNumber(v.Value) }
	end
	if t == "Ray" then
		return { __type = "Ray", Origin = serialize(v.Origin), Direction = serialize(v.Direction) }
	end
	if t == "Rect" then
		return { __type = "Rect", Min = serialize(v.Min), Max = serialize(v.Max) }
	end
	if t == "Font" then
		return { __type = "Font", Family = v.Family, Weight = serialize(v.Weight), Style = serialize(v.Style) }
	end
	return tostring(v)
end

local function deserialize(v)
	if type(v) ~= "table" or not v.__type then return v end
	local t = v.__type
	if t == "nil" then return nil end
	if t == "Vector2" then return Vector2.new(v.X, v.Y) end
	if t == "Vector3" then return Vector3.new(v.X, v.Y, v.Z) end
	if t == "Vector2int16" then return Vector2int16.new(v.X, v.Y) end
	if t == "Vector3int16" then return Vector3int16.new(v.X, v.Y, v.Z) end
	if t == "Color3" then return Color3.new(v.R, v.G, v.B) end
	if t == "UDim" then return UDim.new(v.Scale, v.Offset) end
	if t == "UDim2" then return UDim2.new(v.XScale, v.XOffset, v.YScale, v.YOffset) end
	if t == "BrickColor" then return BrickColor.new(v.Number) end
	if t == "CFrame" then return CFrame.new(table.unpack(v.Components)) end
	if t == "NumberRange" then return NumberRange.new(v.Min, v.Max) end
	if t == "NumberSequence" then
		local points = {}
		for _, p in ipairs(v.Keypoints or {}) do
			points[#points + 1] = NumberSequenceKeypoint.new(p.Time, p.Value, p.Envelope or 0)
		end
		if #points > 0 then return NumberSequence.new(points) end
		return NumberSequence.new(0)
	end
	if t == "ColorSequence" then
		local points = {}
		for _, p in ipairs(v.Keypoints or {}) do
			local c = deserialize(p.Value)
			points[#points + 1] = ColorSequenceKeypoint.new(p.Time, c)
		end
		if #points > 0 then return ColorSequence.new(points) end
		return ColorSequence.new(Color3.new(1, 1, 1))
	end
	if t == "PhysicalProperties" then
		return PhysicalProperties.new(v.Density, v.Friction, v.Elasticity, v.FrictionWeight, v.ElasticityWeight)
	end
	if t == "EnumItem" then
		local enumName = tostring(v.EnumType):match("%.([^%.]+)$") or tostring(v.EnumType)
		if enumName and Enum[enumName] then
			local ok, result = pcall(function()
				return Enum[enumName][v.Name]
			end)
			if ok and result then return result end
			if type(v.Value) == "number" then
				for _, item in ipairs(Enum[enumName]:GetEnumItems()) do
					if item.Value == v.Value then
						return item
					end
				end
			end
		end
		return v.Name
	end
	if t == "Ray" then
		return Ray.new(deserialize(v.Origin), deserialize(v.Direction))
	end
	if t == "Rect" then
		return Rect.new(deserialize(v.Min), deserialize(v.Max))
	end
	if t == "Font" then
		local family = v.Family or "rbxasset://fonts/families/SourceSansPro.json"
		local weight = deserialize(v.Weight) or Enum.FontWeight.Regular
		local style = deserialize(v.Style) or Enum.FontStyle.Normal
		local ok, result = pcall(function()
			return Font.new(family, weight, style)
		end)
		if ok then return result end
	end
	return v
end

local friendly
do
	local function num(v, precise)
		v = tonumber(v)
		if not v then return 0 end
		if v ~= v then return 0 end
		if v == math.huge or v == -math.huge then return 0 end

		local absV = math.abs(v)
		local intEps = math.max(5e-4, absV * 1e-8)
		if intEps > 5e-3 then intEps = 5e-3 end

		local nearestInt = math.floor(v + 0.5)
		if v < 0 then
			nearestInt = math.ceil(v - 0.5)
		end
		if math.abs(v - nearestInt) <= intEps then
			return nearestInt
		end

		if not precise then
			local r
			if v >= 0 then
				r = math.floor(v * 100 + 0.5) / 100
			else
				r = math.ceil(v * 100 - 0.5) / 100
			end
			local ni = math.floor(r + 0.5)
			if r < 0 then
				ni = math.ceil(r - 0.5)
			end
			if math.abs(r - ni) < 1e-9 then
				return ni
			end
			return r
		end

		for decimals = 1, 8 do
			local scale = 10 ^ decimals
			local rounded
			if v >= 0 then
				rounded = math.floor(v * scale + 0.5) / scale
			else
				rounded = math.ceil(v * scale - 0.5) / scale
			end
			local tol = math.max(1e-10, absV * 1e-12)
			if math.abs(v - rounded) <= tol then
				local ni = math.floor(rounded + 0.5)
				if rounded < 0 then
					ni = math.ceil(rounded - 0.5)
				end
				if math.abs(rounded - ni) < 1e-12 then
					return ni
				end
				return rounded
			end
		end

		local s = string.format("%.10g", v)
		local cleaned = tonumber(s)
		if cleaned then
			local ni = math.floor(cleaned + 0.5)
			if cleaned < 0 then
				ni = math.ceil(cleaned - 0.5)
			end
			if math.abs(cleaned - ni) <= intEps then
				return ni
			end
			return cleaned
		end
		return v
	end

	local function vec3(v, precise)
		if typeof(v) ~= "Vector3" then return tostring(v) end
		return string.format("%s, %s, %s", num(v.X, precise), num(v.Y, precise), num(v.Z, precise))
	end

	local function vec2(v, precise)
		if typeof(v) ~= "Vector2" then return tostring(v) end
		return string.format("%s, %s", num(v.X, precise), num(v.Y, precise))
	end

	local function seqPoints(s)
		local points = {}
		if typeof(s) ~= "NumberSequence" then return points end
		for _, kp in ipairs(s.Keypoints) do
			points[#points + 1] = { Time = kp.Time, Value = kp.Value, Envelope = kp.Envelope }
		end
		return points
	end

	local function formatNumberSequence(s, precise)
		local p = seqPoints(s)
		if #p == 0 then
			if type(s) == "number" then
				return "T:0 V:" .. tostring(num(s, precise))
			end
			return tostring(s)
		end
		local out = {}
		for _, k in ipairs(p) do
			local piece = string.format("T:%s V:%s", num(k.Time, precise), num(k.Value, precise))
			local env = tonumber(k.Envelope) or 0
			if env ~= 0 then
				piece = piece .. string.format(" E:%s", num(env, precise))
			end
			out[#out + 1] = piece
		end
		return table.concat(out, "  →  ")
	end

	local function formatColorSequence(s, precise)
		if typeof(s) ~= "ColorSequence" then return tostring(s) end
		local out = {}
		for _, kp in ipairs(s.Keypoints) do
			local c = kp.Value
			local r = math.floor(c.R * 255 + 0.5)
			local g = math.floor(c.G * 255 + 0.5)
			local b = math.floor(c.B * 255 + 0.5)
			out[#out + 1] = string.format("T:%s RGB(%d, %d, %d)", num(kp.Time, precise), r, g, b)
		end
		return table.concat(out, "  →  ")
	end

	local function formatNumberRange(r, precise)
		if typeof(r) ~= "NumberRange" then
			if type(r) == "number" then
				return tostring(num(r, precise))
			end
			return tostring(r)
		end
		local a, b = num(r.Min, precise), num(r.Max, precise)
		if a == b then
			return tostring(a)
		end
		return tostring(a) .. " – " .. tostring(b)
	end

	friendly = function(property, value, precise)
		if value == nil then return "—" end
		precise = precise == true

		local t = typeof(value)

		if t == "boolean" then
			return value and "true" or "false"
		end

		if t == "NumberSequence" then
			return formatNumberSequence(value, precise)
		end
		if t == "ColorSequence" then
			return formatColorSequence(value, precise)
		end
		if t == "NumberRange" then
			return formatNumberRange(value, precise)
		end
		if t == "Vector3" then
			return vec3(value, precise)
		end
		if t == "Vector2" then
			return vec2(value, precise)
		end
		if t == "Color3" then
			local r = math.floor(value.R * 255 + 0.5)
			local g = math.floor(value.G * 255 + 0.5)
			local b = math.floor(value.B * 255 + 0.5)
			return string.format("RGB(%d, %d, %d)", r, g, b)
		end
		if t == "EnumItem" then
			return value.Name
		end
		if t == "number" or type(value) == "number" then
			return tostring(num(value, precise))
		end
		if t == "string" then
			return value
		end

		return tostring(value)
	end
end

local getPath
do
	local PathCache = setmetatable({}, { __mode = "k" })

	getPath = function(o)
		if type(o) == "table" and o.Path then
			return o.Path
		end
		if type(o) ~= "table" and PathCache[o] then
			return PathCache[o]
		end

		local p = {}
		local x = o
		while x and x ~= game do
			p[#p + 1] = x.Name
			x = x.Parent
		end
		for i = 1, math.floor(#p / 2) do
			local j = #p - i + 1
			p[i], p[j] = p[j], p[i]
		end

		local pathStr = table.concat(p, ".")

		if type(o) ~= "table" then
			PathCache[o] = pathStr
		end

		return pathStr
	end
end

local function getTexture(o)
	if type(o) == "table" then
		if o.Texture then return tostring(o.Texture) end
		if o.Properties and o.Properties.Texture then
			local t = o.Properties.Texture
			if type(t) == "table" and t.__type then
				return tostring(deserialize(t) or "")
			end
			return tostring(t or "")
		end
		return ""
	end
	local ok, v = pcall(function()
		return o.Texture
	end)
	return ok and tostring(v or "") or ""
end

local function getEmitCount(o)
	if type(o) == "table" and o.EmitCount then
		return tonumber(o.EmitCount) or 1
	end
	local ok, v = pcall(function()
		return o:GetAttribute("EmitCount")
	end)
	if ok and typeof(v) == "number" then return v end
	return 1
end

local function getInfo(o)
	if type(o) == "table" then
		return o
	end
	local data = {
		Name = tostring(o.Name),
		Class = tostring(o.ClassName),
		Path = getPath(o),
		Texture = getTexture(o),
		EmitCount = getEmitCount(o),
		Properties = {},
		Attributes = {}
	}
	for _, property in ipairs(particleProperties) do
		local ok, v = pcall(function()
			return o[property]
		end)
		if ok and v ~= nil then
			local sOk, sVal = pcall(serialize, v)
			if sOk and sVal ~= nil then
				data.Properties[property] = sVal
			end
		end
	end
	local attrOk, attrs = pcall(function()
		return o:GetAttributes()
	end)
	if attrOk and type(attrs) == "table" then
		for name, value in pairs(attrs) do
			local sOk, sVal = pcall(serialize, value)
			if sOk and sVal ~= nil then
				data.Attributes[tostring(name)] = sVal
			end
		end
	end
	return data
end

-- ===================== SAVE DATABASE (per game) =====================

local saveDatabase
do
	local CHUNK_SIZE = 800

	local function cleanupGameFolder(keepChunkCount)
		for _, path in ipairs(listFolder(GAME_FOLDER)) do
			local name = basename(path)
			if name:match("^particle_%d+%.json$") then
				deleteFile(path)
			else
				local n = tonumber(name:match("^chunk_(%d+)%.json$"))
				if n and n > keepChunkCount then
					deleteFile(path)
				end
			end
		end
	end

	local function saveIndex(index)
		return writeJSON(INDEX_FILE, index)
	end

	local function normalizeEntry(data)
		if type(data) ~= "table" then
			return {
				Name = "Particle",
				Class = "ParticleEmitter",
				Path = "",
				Texture = "",
				EmitCount = 1,
				Properties = {},
				Attributes = {}
			}
		end
		return {
			Name = tostring(data.Name or "Particle"),
			Class = tostring(data.Class or "ParticleEmitter"),
			Path = tostring(data.Path or ""),
			Texture = tostring(data.Texture or ""),
			EmitCount = tonumber(data.EmitCount) or 1,
			Properties = type(data.Properties) == "table" and data.Properties or {},
			Attributes = type(data.Attributes) == "table" and data.Attributes or {}
		}
	end

	saveDatabase = function(particles, statusCallback)
		ensureFolder(ROOT_FOLDER)
		ensureFolder(GAME_FOLDER)

		local count = #particles
		local index = {
			Version = 4,
			GameId = GAME_ID,
			PlaceId = PLACE_ID,
			Count = 0,
			ChunkSize = CHUNK_SIZE,
			Chunks = {},
			Particles = {}
		}

		local savedCount = 0
		local failCount = 0
		local chunkIndex = 0

		local i = 1
		while i <= count do
			local chunk = {}
			local chunkEnd = math.min(i + CHUNK_SIZE - 1, count)

			for j = i, chunkEnd do
				local entry = normalizeEntry(particles[j])
				chunk[#chunk + 1] = entry
			end

			chunkIndex += 1
			local filename = "chunk_" .. string.format("%04d", chunkIndex) .. ".json"
			local path = joinPath(GAME_FOLDER, filename)

			local payload = { Particles = chunk }
			local ok = writeJSON(path, payload)

			if not ok then
				local stripped = {}
				for _, e in ipairs(chunk) do
					stripped[#stripped + 1] = {
						Name = e.Name,
						Class = e.Class,
						Path = e.Path,
						Texture = e.Texture,
						EmitCount = e.EmitCount,
						Properties = e.Properties,
						Attributes = {}
					}
				end
				ok = writeJSON(path, { Particles = stripped })
				if ok then
					chunk = stripped
				end
			end

			if ok then
				local n = #chunk
				savedCount += n
				index.Chunks[chunkIndex] = {
					File = filename,
					Count = n
				}
			else
				failCount += #chunk
				warn("[ParticleInspector] Failed to save chunk #" .. tostring(chunkIndex))
			end

			if statusCallback then
				statusCallback(chunkEnd, count)
			end
			task.wait()

			i = chunkEnd + 1
		end

		index.Count = savedCount
		cleanupGameFolder(chunkIndex)

		local ok, err = saveIndex(index)
		if not ok then return false, err, savedCount, failCount end
		return true, nil, savedCount, failCount
	end
end

				Count = 0,
				Particles = {}
			}
		end
		index.Particles = index.Particles or {}
		index.Count = tonumber(index.Count) or 0
		return index
	end

	local function saveFavIndex(index)
		return writeJSON(FAV_INDEX_FILE, index)
	end

	local function cleanupOldFavFiles(keepCount)
		for _, path in ipairs(listFolder(FAV_FOLDER)) do
			local name = basename(path)
			local n = tonumber(name:match("^fav_(%d+)%.json$"))
			if n and n > keepCount then
				deleteFile(path)
			end
		end
	end

	local function saveFavParticleFile(index, data)
		local path = favParticleFile(index)
		local ok, err = writeJSON(path, data)
		if ok then return true end
		if type(data) == "table" and data.Attributes then
			local stripped = {}
			for k, v in pairs(data) do
				if k ~= "Attributes" then
					stripped[k] = v
				end
			end
			ok, err = writeJSON(path, stripped)
			if ok then return true end
		end
		local minimal = {
			Name = tostring(data and data.Name or "Particle"),
			Class = tostring(data and data.Class or "ParticleEmitter"),
			Path = tostring(data and data.Path or ""),
			Texture = tostring(data and data.Texture or ""),
			EmitCount = tonumber(data and data.EmitCount) or 1,
			Properties = (type(data) == "table" and type(data.Properties) == "table") and data.Properties or {},
			Attributes = {}
		}
		ok, err = writeJSON(path, minimal)
		if ok then return true end
		return false, err or "save failed"
	end

	local function makeFavKey(data)
		if type(data) ~= "table" then return nil end
		local path = tostring(data.Path or "")
		local name = tostring(data.Name or "")
		local tex = tostring(data.Texture or "")
		return path .. "|" .. name .. "|" .. tex
	end

	rebuildFavouritesList = function()
		FavouritesList = {}
		for _, data in pairs(FavouritesMap) do
			FavouritesList[#FavouritesList + 1] = data
		end
		table.sort(FavouritesList, function(a, b)
			local pa = tostring(a.Path or ""):lower()
			local pb = tostring(b.Path or ""):lower()
			if pa == pb then
				return tostring(a.Name or ""):lower() < tostring(b.Name or ""):lower()
			end
			return pa < pb
		end)
	end

	loadFavouritesFromDisk = function()
		ensureFolder(FAV_FOLDER)
		table.clear(FavouritesMap)
		local idx = loadFavIndex()
		local particles = idx.Particles or {}
		local entries = {}
		for _, v in pairs(particles) do
			if type(v) == "table" and v.File then
				entries[#entries + 1] = v
			end
		end
		for _, pData in ipairs(entries) do
			local filePath = joinPath(FAV_FOLDER, pData.File)
			local fullData = loadParticleFile(filePath)
			if fullData then
				local key = makeFavKey(fullData)
				if key then
					FavouritesMap[key] = fullData
				end
			end
		end
		rebuildFavouritesList()
	end

	local function persistFavourites()
		ensureFolder(ROOT_FOLDER)
		ensureFolder(FAV_FOLDER)
		rebuildFavouritesList()
		local index = {
			Version = 1,
			Count = 0,
			Particles = {}
		}
		local savedCount = 0
		for _, data in ipairs(FavouritesList) do
			local entry = data
			if type(entry) ~= "table" then
				entry = {
					Name = "Particle",
					Class = "ParticleEmitter",
					Path = "",
					Texture = "",
					EmitCount = 1,
					Properties = {},
					Attributes = {}
				}
			end
			local filename = "fav_" .. string.format("%06d", savedCount + 1) .. ".json"
			local ok = saveFavParticleFile(savedCount + 1, entry)
			if ok then
				savedCount = savedCount + 1
				index.Particles[savedCount] = {
					File = filename,
					Name = tostring(entry.Name or "Particle"),
					Class = tostring(entry.Class or "ParticleEmitter"),
					Path = tostring(entry.Path or ""),
					EmitCount = tonumber(entry.EmitCount) or 1
				}
			end
		end
		index.Count = savedCount
		cleanupOldFavFiles(savedCount)
		saveFavIndex(index)
		return savedCount
	end

	isFavourited = function(o)
		local data = getInfo(o)
		local key = makeFavKey(data)
		return key and FavouritesMap[key] ~= nil
	end

	local function addFavourite(o)
		local data = getInfo(o)
		if type(data) ~= "table" then return false end
		local snapshot = {
			Name = tostring(data.Name or "Particle"),
			Class = tostring(data.Class or "ParticleEmitter"),
			Path = tostring(data.Path or ""),
			Texture = tostring(data.Texture or ""),
			EmitCount = tonumber(data.EmitCount) or 1,
			Properties = type(data.Properties) == "table" and data.Properties or {},
			Attributes = type(data.Attributes) == "table" and data.Attributes or {}
		}
		local key = makeFavKey(snapshot)
		if not key then return false end
		if FavouritesMap[key] then return true end
		FavouritesMap[key] = snapshot
		rebuildFavouritesList()
		persistFavourites()
		return true
	end

	removeFavourite = function(o)
		local data = getInfo(o)
		local key = makeFavKey(data)
		if not key or not FavouritesMap[key] then return false end
		FavouritesMap[key] = nil
		rebuildFavouritesList()
		persistFavourites()
		return true
	end

	toggleFavourite = function(o)
		if isFavourited(o) then
			return removeFavourite(o), false
		else
			return addFavourite(o), true
		end
	end
end

function Metal.shade(c, f)
	return Color3.new(
		math.clamp(c.R * f, 0, 1),
		math.clamp(c.G * f, 0, 1),
		math.clamp(c.B * f, 0, 1)
	)
end

-- Gradient straight on a frame (only for things WITHOUT their own text,
-- because a UIGradient also tints text on the same object).
function Metal.panel(inst, style, strength)
	if not inst then return nil end
	strength = strength or 1
	local old = inst:FindFirstChild("MetalGradient")
	if old then old:Destroy() end

	local points = Metal.RAISED
	if style == "inset" then
		points = Metal.INSET
	elseif style == "bar" then
		points = Metal.BAR
	end

	local keypoints = {}
	for _, p in ipairs(points) do
		local f = 1 - (1 - p[2]) * strength
		keypoints[#keypoints + 1] = ColorSequenceKeypoint.new(p[1], Color3.new(f, f, f))
	end

	local g = Instance.new("UIGradient")
	g.Name = "MetalGradient"
	g.Rotation = 90
	g.Color = ColorSequence.new(keypoints)
	g.Parent = inst
	return g
end

-- Glossy chrome overlay for buttons that have text (bright top, dark horizon, light bounce at the bottom).
-- It never blocks clicks.
function Metal.gloss(btn, radius, intensity)
	if not btn then return nil end
	local existing = btn:FindFirstChild("MetalGloss")
	if existing then return existing end
	intensity = intensity or 1

	local g = Instance.new("Frame")
	g.Name = "MetalGloss"
	g.Size = UDim2.fromScale(1, 1)
	g.Position = UDim2.fromScale(0, 0)

	g.BackgroundTransparency = 0
	g.BorderSizePixel = 0
	g.Active = false

	g.Parent = btn

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, radius or 5)
	corner.Parent = g

	local function t(v)
		return math.clamp(1 - (1 - v) * intensity, 0, 1)

	local grad = Instance.new("UIGradient")

	grad.Color = ColorSequence.new({

-- ===================== ANIMATION HELPERS =====================

local Anim = {}

local SKIP_FADE = {
	HoverStroke = true,
	PulseStroke = true,
	FocusStroke = true,

function Anim.tween(inst, time, props, style, dir)
	if not inst then return nil end
	local ok, tw = pcall(function()
		local t = TweenService:Create(
			inst,
			TweenInfo.new(time or 0.2, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
			props
		)
		t:Play()
		return t
	end)
	if ok then return tw end
	return nil
end

-- Smooth background colour change
local function setBg(inst, color, t)
	Anim.tween(inst, t or 0.18, { BackgroundColor3 = color })
end

local function transparencyProps(o, useCurrent)
	local p = nil
	if o:IsA("GuiObject") then
		p = { BackgroundTransparency = useCurrent and o.BackgroundTransparency or 1 }
		if o:IsA("TextLabel") or o:IsA("TextButton") or o:IsA("TextBox") then
			p.TextTransparency = useCurrent and o.TextTransparency or 1
		end
		if o:IsA("ImageLabel") or o:IsA("ImageButton") then
			p.ImageTransparency = useCurrent and o.ImageTransparency or 1
		end
	elseif o:IsA("UIStroke") then
		p = { Transparency = useCurrent and o.Transparency or 1 }
	end
	return p
end

-- Fades a whole piece of GUI in (everything starts invisible, then appears)
function Anim.fadeIn(root, delay, duration)
	if not root then return end
	local list = {}
	local function add(o)
		if SKIP_FADE[o.Name] then return end
		local target = transparencyProps(o, true)
		if target then
			list[#list + 1] = { o, target }
			for k in pairs(target) do
				o[k] = 1
			end
		end
	end
	add(root)
	for _, d in ipairs(root:GetDescendants()) do
		add(d)
	end
	task.delay(delay or 0, function()
		for _, item in ipairs(list) do
			local o, target = item[1], item[2]
			if o.Parent then
				Anim.tween(o, duration or 0.28, target)
			end
		end
	end)
end

-- Fades a whole piece of GUI out
function Anim.fadeOut(root, duration)
	if not root then return end
	local function go(o)
		if SKIP_FADE[o.Name] then return end
		local target = transparencyProps(o, false)
		if target then
			Anim.tween(o, duration or 0.25, target)
		end
	end
	go(root)
	for _, d in ipairs(root:GetDescendants()) do
		go(d)
	end
end

-- Hover glow + press feedback for buttons
function Anim.button(btn, opts)
	if not btn then return nil end
	if btn:FindFirstChild("HoverStroke") then return nil end
	opts = opts or {}
	local hoverT = opts.hoverTransparency or 0.35

	local stroke = Instance.new("UIStroke")
	stroke.Name = "HoverStroke"
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Color = opts.color or Color3.fromRGB(235, 240, 255)
	stroke.Thickness = 1.2
	stroke.Transparency = 1
	stroke.Parent = btn

	local hovering = false

	btn.MouseEnter:Connect(function()
		hovering = true
		Anim.tween(stroke, 0.15, { Transparency = hoverT })
	end)
	btn.MouseLeave:Connect(function()
		hovering = false
		Anim.tween(stroke, 0.2, { Transparency = 1, Thickness = 1.2 })
	end)
	btn.MouseButton1Down:Connect(function()
		Anim.tween(stroke, 0.07, { Thickness = 2.6, Transparency = 0 })
	end)
	btn.MouseButton1Up:Connect(function()
		Anim.tween(stroke, 0.2, { Thickness = 1.2, Transparency = hovering and hoverT or 1 }, Enum.EasingStyle.Back)
	end)
	return stroke
end

-- Light hover colour fade (used on very long lists)
function Anim.rowHover(row, baseColor, hoverColor)
	row.MouseEnter:Connect(function()
		Anim.tween(row, 0.12, { BackgroundColor3 = hoverColor })
	end)
	row.MouseLeave:Connect(function()
		Anim.tween(row, 0.18, { BackgroundColor3 = baseColor })
	end)
end

-- Quick side-to-side shake
function Anim.shake(inst)
	task.spawn(function()
		local base = inst.Position
		for _, dx in ipairs({ 10, -10, 7, -5, 3, 0 }) do
			if not inst.Parent then return end
			Anim.tween(inst, 0.05, { Position = base + UDim2.fromOffset(dx, 0) })
			task.wait(0.05)
		end
	end)
end

local function isValidParticle(o)
	if not o then return false end
	if type(o) == "table" then
		return o.Name ~= nil or o.Properties ~= nil
	end
	local ok, isEmitter = pcall(function()
		return o:IsA("ParticleEmitter")
	end)
	return ok and isEmitter and o.Parent ~= nil
end

local function stopPreview()
	Previewing = false
	if PreviewPart then
		PreviewPart:Destroy()
		PreviewPart = nil
	end

end

local function applyPropertiesToEmitter(emitter, sourceProps, isLive)
	local applied = {}

	local function setProp(name, value)
		if value == nil then return end
		local ok = pcall(function()
			emitter[name] = value
		end)
		if ok then
			applied[name] = true
		end
	end

	if isLive then
		local src = sourceProps
		pcall(function() emitter.Texture = src.Texture end)
		applied.Texture = true
		pcall(function() emitter.FlipbookLayout = src.FlipbookLayout end)
		applied.FlipbookLayout = true
		pcall(function() emitter.FlipbookSizeX = src.FlipbookSizeX end)
		pcall(function() emitter.FlipbookSizeY = src.FlipbookSizeY end)
		applied.FlipbookSizeX = true
		applied.FlipbookSizeY = true
		pcall(function() emitter.FlipbookBlendFrames = src.FlipbookBlendFrames end)
		pcall(function() emitter.FlipbookMode = src.FlipbookMode end)
		pcall(function() emitter.FlipbookFramerate = src.FlipbookFramerate end)
		pcall(function() emitter.FlipbookStartRandom = src.FlipbookStartRandom end)
		applied.FlipbookBlendFrames = true
		applied.FlipbookMode = true
		applied.FlipbookFramerate = true
		applied.FlipbookStartRandom = true
		for _, prop in ipairs(particleProperties) do
			if not applied[prop] then
				pcall(function()
					emitter[prop] = src[prop]
				end)
			end
		end
	else
		local props = sourceProps
		local function getDeserialized(name)
			local raw = props[name]
			if raw == nil then return nil end
			return deserialize(raw)
		end
		setProp("Texture", getDeserialized("Texture"))
		setProp("FlipbookLayout", getDeserialized("FlipbookLayout"))
		local sx = getDeserialized("FlipbookSizeX")
		local sy = getDeserialized("FlipbookSizeY")
		if sx ~= nil then
			setProp("FlipbookSizeX", math.clamp(math.floor(tonumber(sx) or 1), 1, 64))
		end
		if sy ~= nil then
			setProp("FlipbookSizeY", math.clamp(math.floor(tonumber(sy) or 1), 1, 64))
		end
		setProp("FlipbookBlendFrames", getDeserialized("FlipbookBlendFrames"))
		setProp("FlipbookMode", getDeserialized("FlipbookMode"))
		setProp("FlipbookFramerate", getDeserialized("FlipbookFramerate"))
		setProp("FlipbookStartRandom", getDeserialized("FlipbookStartRandom"))
		for prop, val in pairs(props) do
			if not applied[prop] then
				setProp(prop, deserialize(val))
			end
		end
	end

	emitter.Enabled = true
end

local function startPreview(target)
	stopPreview()
	if not target then return end

	local spawnCFrame
	if LP and LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") then
		local hrp = LP.Character.HumanoidRootPart
		spawnCFrame = hrp.CFrame * CFrame.new(0, 0, -7)
	elseif workspace.CurrentCamera then
		spawnCFrame = workspace.CurrentCamera.CFrame * CFrame.new(0, 0, -7)
	else
		return
	end

	PreviewPart = Instance.new("Part")
	PreviewPart.Name = "ParticlePreviewPart"
	PreviewPart.Size = Vector3.new(0.5, 0.5, 0.5)
	PreviewPart.Transparency = 1
	PreviewPart.CanCollide = false
	PreviewPart.Anchored = true
	PreviewPart.CFrame = spawnCFrame
	PreviewPart.Parent = workspace

	PreviewEmitter = Instance.new("ParticleEmitter")

	if type(target) ~= "table" and typeof(target) == "Instance" and target:IsA("ParticleEmitter") then
		applyPropertiesToEmitter(PreviewEmitter, target, true)
	elseif type(target) == "table" and target.Properties then
		applyPropertiesToEmitter(PreviewEmitter, target.Properties, false)
	else
		PreviewPart:Destroy()
		PreviewPart = nil
		return
	end

	PreviewEmitter.Parent = PreviewPart

	local emitCount = getEmitCount(target)
	emitCount = math.clamp(math.floor(tonumber(emitCount) or 1), 1, 10000)
	pcall(function()
		PreviewEmitter.Enabled = true
		PreviewEmitter:Emit(emitCount)
	end)

	Previewing = true
	PreviewButton.Text = "STOP"
	setBg(PreviewButton, C.red)
end

local function measureTextHeight(text, textSize, font, maxWidth)
	local ok, size = pcall(function()
		return TextService:GetTextSize(
			tostring(text),
			textSize,
			font,
			Vector2.new(math.max(maxWidth, 20), 10000)
		)
	end)
	if ok and size then
		return math.max(size.Y, textSize + 2)
	end
	local charsPerLine = math.max(1, math.floor(maxWidth / (textSize * 0.45)))
	local lines = math.ceil(#tostring(text) / charsPerLine)
	return math.max(lines * (textSize + 2), textSize + 2)
end

local function copyToClipboard(text)
	text = tostring(text or "")
	if text == "" then return false end
	if type(setclipboard) == "function" then
		local ok = pcall(setclipboard, text)
		if ok then return true end
	end
	if type(toclipboard) == "function" then
		local ok = pcall(toclipboard, text)
		if ok then return true end
	end
	if type(Clipboard) == "table" and type(Clipboard.set) == "function" then
		local ok = pcall(Clipboard.set, text)
		if ok then return true end
	end
	return false
end

local function showInfo(o, keepPreview, noAnim)
	if not isValidParticle(o) then return end
	Selected = o
	if Previewing and not keepPreview then stopPreview() end

	InfoTitle.Text = (type(o) == "table" and o.Name) or o.Name or "Particle"
	if not noAnim then
		InfoTitle.TextTransparency = 1
		Anim.tween(InfoTitle, 0.3, { TextTransparency = 0 })
	end
	clearInfo()
	updateFavButton()

	InfoAnimOn = not noAnim
	InfoAnimIndex = 0

	local texture = getTexture(o)
	local preview = Instance.new("Frame")
	preview.Size = UDim2.new(1, -2, 0, 70)
	preview.BackgroundColor3 = Color3.fromRGB(10, 10, 14)
	preview.BorderSizePixel = 0
	preview.Parent = InfoScroll
	Instance.new("UICorner", preview).CornerRadius = UDim.new(0, 6)

	if texture ~= "" then
		local image = Instance.new("ImageLabel")
		image.Size = UDim2.new(1, -10, 1, -10)
		image.Position = UDim2.fromOffset(5, 5)
		image.BackgroundTransparency = 1
		image.Image = texture
		image.ScaleType = Enum.ScaleType.Fit
		image.Parent = preview
	else
		local txt = Instance.new("TextLabel")
		txt.Size = UDim2.fromScale(1, 1)
		txt.BackgroundTransparency = 1
		txt.Text = "NO TEXTURE"
		txt.TextColor3 = C.textMuted
		txt.Font = Enum.Font.GothamBold
		txt.TextSize = 10
		txt.Parent = preview
	end

	if InfoAnimOn then
		Anim.fadeIn(preview, 0, 0.3)
	end

	local d = getInfo(o)
	local shown = {}

	local function markShown(label)
		shown[string.lower(tostring(label))] = true
	end

	local function alreadyShown(label)
		return shown[string.lower(tostring(label))] == true
	end

	infoLine("CLASS", d.Class or "ParticleEmitter")
	markShown("CLASS")
	infoLine("PATH", d.Path or "")
	markShown("PATH")
	infoLine("EMIT", tostring(d.EmitCount or 1) .. " particles")
	markShown("EMIT")
	do
		local texVal = (d.Texture and d.Texture ~= "") and d.Texture or "None"
		infoLine("TEXTURE", texVal, { copyText = (texVal ~= "None") and texVal or nil })
	end
	markShown("TEXTURE")

	if d.Properties then
		for _, p in ipairs(particleProperties) do
			if p == "FlipbookSizeX" or p == "FlipbookSizeY" then
			elseif INFO_HEADER_PROPS[p] then
			elseif alreadyShown(p) then
			elseif d.Properties[p] ~= nil then
				local raw = deserialize(d.Properties[p])
				local display = friendly(p, raw, false)
				local precise = friendly(p, raw, true)
				if p == "FlipbookLayout" then
					local layoutVal = raw
					local isCustom = false
					if typeof(layoutVal) == "EnumItem" then
						isCustom = (layoutVal.Name == "Custom")
					elseif type(layoutVal) == "string" then
						isCustom = (layoutVal == "Custom")
					end
					if isCustom then
						local sx = tonumber(deserialize(d.Properties.FlipbookSizeX)) or 1
						local sy = tonumber(deserialize(d.Properties.FlipbookSizeY)) or 1
						sx = math.floor(sx)
						sy = math.floor(sy)
						display = string.format("Custom (%dx%d)", sx, sy)
						precise = display
					end
				end
				infoLine(p, display, { preciseValue = precise })
				markShown(p)
			end
		end
	end

	if d.Attributes then
		for n, v in pairs(d.Attributes) do
			if not alreadyShown(n) then
				local raw = deserialize(v)
				local display = friendly(n, raw, false)
				local precise = friendly(n, raw, true)
				infoLine(n, display, { preciseValue = precise })
				markShown(n)
			end
		end
	end

	InfoAnimOn = false
end

rebuildFilter = function(forceYield)
	if FilterBusy then return end
	FilterBusy = true

	local result = {}
	local q = Search.Text:lower()
	local listLen = #ParticleList
	local shouldYield = forceYield or listLen > 6000

	if q == "" then
		for i = 1, listLen do
			local o = ParticleList[i]
			if isValidParticle(o) then
				result[#result + 1] = o
			else
				if type(o) ~= "table" then
					ParticleMap[o] = nil
				end
			end
			if shouldYield and i % 5000 == 0 then
				Status.Text = "Filter " .. i
				task.wait()
			end
		end
	else
		local sortData = {}
		for i = 1, listLen do
			local o = ParticleList[i]
			if isValidParticle(o) then
				local n = (type(o) == "table" and o.Name) or o.Name or "Particle"
				local p = getPath(o)
				local nLower = n:lower()
				local pLower = p:lower()
				if nLower:find(q, 1, true) or pLower:find(q, 1, true) then
					result[#result + 1] = o
					sortData[o] = { name = nLower, path = pLower }
				end
			else
				if type(o) ~= "table" then
					ParticleMap[o] = nil
				end
			end
			if shouldYield and i % 4000 == 0 then
				Status.Text = "Filter " .. i
				task.wait()
			end
		end
		if #result > 1 then
			table.sort(result, function(a, b)
				local aData = sortData[a]
				local bData = sortData[b]
				if aData.path == bData.path then
					return aData.name < bData.name
				end
				return aData.path < bData.path
			end)
		end
	end

	Filtered = result
	local height = math.max(#Filtered * RowHeight, ListViewH + 1)
	if height > 5000000 then
		height = 5000000
	end
	ListContent.Size = UDim2.new(1, -4, 0, height)
	List.CanvasSize = UDim2.fromOffset(0, height)

	Status.Text = tostring(#Filtered) .. "/" .. tostring(#ParticleList)

	LastFirst = -1
	LastLast = -1
	FilterBusy = false
end

local function updateRows()
	local first = math.max(1, math.floor(List.CanvasPosition.Y / RowHeight) - Overscan + 1)
	local visible = math.ceil(ListViewH / RowHeight)
	local last = math.min(#Filtered, first + visible + Overscan * 2)

	if first == LastFirst and last == LastLast then
		updateSelectionVisuals()
		return
	end

	LastFirst = first
	LastLast = last

	for i, row in ipairs(Rows) do
		local index = first + i - 1
		if index <= last then
			local o = Filtered[index]
			if o and isValidParticle(o) then
				setRow(row, index, o)
			else
				row.Visible = false
				RowData[row] = nil
			end
		else
			row.Visible = false
			RowData[row] = nil
		end
	end
end

List:GetPropertyChangedSignal("CanvasPosition"):Connect(updateRows)

local function addParticle(o)
	if not o then return false end
	if #ParticleList >= MAX_PARTICLES then
		return false
	end
	if type(o) == "table" then
		if not ParticleMap[o] then
			ParticleMap[o] = true
			ParticleList[#ParticleList + 1] = o
			return true
		end
		return false
	end
	if o.Parent and o:IsA("ParticleEmitter") then
		if not ParticleMap[o] then
			ParticleMap[o] = true
			ParticleList[#ParticleList + 1] = o
			return true
		end
	end
	return false
end

-- AUTO SCAN PIPELINE
local AutoQueue = {}
local AutoQueueHead = 1
local AutoQueueTail = 1
local AutoSeenSet = setmetatable({}, { __mode = "k" })
local AutoDirty = false

workspace.DescendantAdded:Connect(function(descendant)
	if not AutoScanning then return end
	if descendant:IsA("ParticleEmitter") then
		if not AutoSeenSet[descendant] then
			AutoSeenSet[descendant] = true
			local queued = AutoQueueTail - AutoQueueHead
			if queued >= MAX_AUTO_QUEUE then
				AutoQueue[AutoQueueHead] = nil
				AutoQueueHead += 1
			end
			AutoQueue[AutoQueueTail] = descendant
			AutoQueueTail += 1
		end
	end
end)

do
	local TIME_BETWEEN_UI_UPDATES = 0.4
	local lastUIUpdate = os.clock()
	local MAX_PROCESS_PER_FRAME = 3
	local AUTO_TIME_BUDGET = 0.0012

	RunService.Heartbeat:Connect(function()
		if not AutoScanning then return end

		local startTime = os.clock()
		local processed = 0
		local atAutoCap = AutoScannedCount >= MAX_AUTO_CAPTURES

		while AutoQueueHead < AutoQueueTail
			and processed < MAX_PROCESS_PER_FRAME
			and (os.clock() - startTime) < AUTO_TIME_BUDGET
		do
			local emitter = AutoQueue[AutoQueueHead]
			AutoQueue[AutoQueueHead] = nil
			AutoQueueHead += 1
			processed += 1

			if atAutoCap then
			elseif emitter and emitter.Parent and emitter:IsA("ParticleEmitter") then
				local ok, snapshot = pcall(getInfo, emitter)
				if ok and type(snapshot) == "table" then
					if not AutoScannedSet[snapshot] then
						AutoScannedSet[snapshot] = true
						AutoScannedList[#AutoScannedList + 1] = snapshot
						AutoScannedCount = #AutoScannedList
						atAutoCap = AutoScannedCount >= MAX_AUTO_CAPTURES
						AutoDirty = true
					end
				end
			end
		end

		if AutoQueueHead == AutoQueueTail then
			if AutoQueueHead > 500 then
				AutoQueue = {}
				AutoQueueHead = 1
				AutoQueueTail = 1
			else
				AutoQueueHead = 1
				AutoQueueTail = 1
			end
		end

		if AutoDirty and (os.clock() - lastUIUpdate) >= TIME_BETWEEN_UI_UPDATES then
			AutoDirty = false
			lastUIUpdate = os.clock()
			Status.Text = tostring(#Filtered) .. "/" .. tostring(#ParticleList)
			if AutoPage.Visible then
				renderAutoPage()
			end
		end
	end)
end

local scan
do
	local SCAN_YIELD_EVERY = 120
	local SCAN_STATUS_EVERY = 400

	local function shouldSkipInstance(inst)
		if not inst then return true end
		local name = inst.Name
		if inst == CoreGui then return true end
		if name == "CoreGui" or name == "Chat" or name == "BubbleChat" then return true end
		local className = inst.ClassName
		if className == "CoreGui" or className == "PlayerGui" then return true end
		return false
	end

	scan = function()
		if ScanBusy then return end
		ScanBusy = true
		Refresh.Text = "..."
		setBg(Refresh, C.orange)
		Status.Text = "Scanning..."

		task.spawn(function()
			local found = 0
			local visited = 0
			local stack = {}

			local roots = { workspace }
			pcall(function()
				table.insert(roots, game:GetService("ReplicatedStorage"))
			end)
			pcall(function()
				table.insert(roots, game:GetService("ReplicatedFirst"))
			end)
			pcall(function()
				table.insert(roots, Players)
			end)
			pcall(function()
				table.insert(roots, game:GetService("Lighting"))
			end)
			pcall(function()
				table.insert(roots, game:GetService("StarterGui"))
			end)
			pcall(function()
				table.insert(roots, game:GetService("StarterPack"))
			end)
			pcall(function()
				table.insert(roots, game:GetService("StarterPlayer"))
			end)

			for _, root in ipairs(roots) do
				if root then
					stack[#stack + 1] = root
				end
			end

			while #stack > 0 do
				local inst = table.remove(stack)
				visited += 1

				if not shouldSkipInstance(inst) then
					local isEmitter = false
					local ok = pcall(function()
						isEmitter = inst:IsA("ParticleEmitter")
					end)
					if ok and isEmitter then
						if addParticle(inst) then
							found += 1
						end
					end

					local childrenOk, children = pcall(function()
						return inst:GetChildren()
					end)
					if childrenOk and type(children) == "table" then
						for i = 1, #children do
							stack[#stack + 1] = children[i]
						end
					end
				end

				if visited % SCAN_YIELD_EVERY == 0 then
					if visited % SCAN_STATUS_EVERY == 0 then
						Status.Text = "S " .. tostring(found)
					end
					task.wait()
				end
			end

			Status.Text = "Filter..."
			rebuildFilter(true)
			updateRows()

			ScanBusy = false
			if Refresh.Parent then
				Refresh.Text = "Refresh"
				setBg(Refresh, C.surface3)
			end
			Status.Text = tostring(#Filtered) .. "/" .. tostring(#ParticleList)
		end)
	end
end

local saving = false
local function saveCurrent()
	if saving then return false, "Already saving" end
	saving = true
	Save.Text = "SAVING..."
	setBg(Save, C.orange)

	local particles = {}

	for i = 1, #ParticleList do
		local o = ParticleList[i]
		if isValidParticle(o) then
			local ok, info = pcall(getInfo, o)
			if ok and type(info) == "table" then
				particles[#particles + 1] = info
			else
				particles[#particles + 1] = {
					Name = (type(o) == "table" and o.Name) or (typeof(o) == "Instance" and o.Name) or "Particle",
					Class = (type(o) == "table" and o.Class) or "ParticleEmitter",
					Path = (type(o) == "table" and o.Path) or getPath(o) or "",
					Texture = getTexture(o),
					EmitCount = getEmitCount(o),
					Properties = (type(o) == "table" and type(o.Properties) == "table") and o.Properties or {},
					Attributes = (type(o) == "table" and type(o.Attributes) == "table") and o.Attributes or {}
				}
			end
		elseif type(o) == "table" then
			particles[#particles + 1] = o
		end
		if i % 200 == 0 then
			Status.Text = "Prep " .. i
			task.wait()
		end
	end

	local totalToSave = #particles
	Status.Text = "0/" .. tostring(totalToSave)

	local ok, err, savedCount, failCount = saveDatabase(particles, function(done, total)
		Save.Text = math.floor(done / math.max(total, 1) * 100) .. "%"
		Status.Text = tostring(done) .. "/" .. tostring(total)
	end)
	saving = false

	savedCount = savedCount or 0
	failCount = failCount or 0

	if not ok then
		warn("[ParticleInspector] Save failed:", err)
		Save.Text = "FAILED"
		setBg(Save, C.red)
		Status.Text = tostring(savedCount) .. "/" .. tostring(totalToSave)
	else
		if failCount > 0 then
			Save.Text = savedCount .. "/" .. totalToSave
			setBg(Save, C.orange)
			warn("[ParticleInspector] Saved", savedCount, "of", totalToSave, "| failed:", failCount)
		else
			Save.Text = "SAVED " .. tostring(savedCount)
			setBg(Save, C.green)
		end
		Status.Text = tostring(savedCount) .. "/" .. tostring(totalToSave)
	end

	task.delay(2.2, function()
		if Save.Parent then
			Save.Text = "SAVE"
			setBg(Save, C.accent, 0.3)
		end
	end)
	return ok
end

local loadingPack = false

local function openSavedPack(gameFolderPath)
	if loadingPack then return end

	local idx = readJSON(joinPath(gameFolderPath, "index.json"))
	if type(idx) ~= "table" then
		Status.Text = "No index"
		return
	end

	local version = tonumber(idx.Version) or 3
	local isChunked = (version >= 4) and type(idx.Chunks) == "table" and next(idx.Chunks) ~= nil

	loadingPack = true

	table.clear(ParticleMap)
	table.clear(ParticleList)
	table.clear(AutoScannedSet)
	table.clear(AutoScannedList)
	AutoScannedCount = 0
	clearAutoList()
	Selected = nil
	clearInfo()
	InfoTitle.Text = "Select a particle"
	stopPreview()
	updateFavButton()
	LastFirst = -1
	LastLast = -1

	task.spawn(function()
		local loaded = 0

		if isChunked then
			local chunks = {}
			for k, v in pairs(idx.Chunks) do
				if type(v) == "table" and v.File then
					local order = tonumber(k) or (#chunks + 1)
					chunks[#chunks + 1] = { Order = order, File = v.File, Count = tonumber(v.Count) or 0 }
				end
			end
			table.sort(chunks, function(a, b)
				return a.Order < b.Order
			end)

			local totalChunks = #chunks
			if totalChunks == 0 then
				Status.Text = "0 particles"
				loadingPack = false
				return
			end

			for ci, cInfo in ipairs(chunks) do
				local filePath = joinPath(gameFolderPath, cInfo.File)
				local chunkData = loadParticleFile(filePath)
				local list = nil
				if type(chunkData) == "table" then
					if type(chunkData.Particles) == "table" then
						list = chunkData.Particles
					elseif #chunkData > 0 then
						list = chunkData
					end
				end
				if type(list) == "table" then
					for _, fullData in ipairs(list) do
						if type(fullData) == "table" then
							if addParticle(fullData) then
								loaded += 1
							end
						end
					end
				end
				Status.Text = "Load " .. tostring(ci) .. "/" .. tostring(totalChunks)
				task.wait()
			end
		else
			local particles = idx.Particles
			if type(particles) ~= "table" then
				Status.Text = "Empty pack"
				loadingPack = false
				return
			end

			local entries = {}
			for k, v in pairs(particles) do
				if type(v) == "table" and v.File then
					entries[#entries + 1] = v
				end
			end

			if #entries == 0 then
				Status.Text = "0 particles"
				loadingPack = false
				return
			end

			local total = #entries
			for i, pData in ipairs(entries) do
				local filePath = joinPath(gameFolderPath, pData.File)
				local fullData = loadParticleFile(filePath)
				if fullData then
					if addParticle(fullData) then
						loaded += 1
					end
				end
				if i % 40 == 0 then
					Status.Text = "Loading " .. tostring(i) .. "/" .. tostring(total)
					task.wait()
				end
			end
		end

		setPage(1)
		rebuildFilter(true)
		updateRows()
		Status.Text = tostring(#Filtered) .. "/" .. tostring(loaded)
		loadingPack = false
	end)
end

local tabPositions = { 0, 0.25, 0.5, 0.75 }

-- FIX: one place that stops every running page slide and puts every page exactly where
-- the layout says it belongs. This is what stops pages from ending up in the wrong spot.
local function snapPages()
	for page, tw in pairs(PageTweens) do
		pcall(function() tw:Cancel() end)
		PageTweens[page] = nil
	end
	for _, pg in ipairs({ ParticlePage, AutoPage, FavPage, SavedPage }) do
		pg.Position = UDim2.fromOffset(0, pageTopValue)
	end
end

-- instant = true  -> no slide animation (used by info-only mode and un-minimize,
--                    where the whole window is being rebuilt anyway)
setPage = function(pageIndex, instant)
	local prevPage = currentPage
	currentPage = pageIndex

	-- always start from clean, correct positions
	snapPages()

	ParticlePage.Visible = (pageIndex == 1)
	AutoPage.Visible = (pageIndex == 2)
	FavPage.Visible = (pageIndex == 3)
	SavedPage.Visible = (pageIndex == 4)

	-- tab text colours + sliding highlight
	local tabs = { ParticlePageButton, AutoPageButton, FavPageButton, SavedPageButton }
	for i, tab in ipairs(tabs) do
		if instant then
			tab.TextColor3 = (i == pageIndex) and C.text or C.tabIdle
		else
			Anim.tween(tab, 0.22, { TextColor3 = (i == pageIndex) and C.text or C.tabIdle })
		end
	end
	local indicatorPos = UDim2.new(tabPositions[pageIndex] or 0, 1, 0, 2)
	if instant then
		TabIndicator.Position = indicatorPos
	else
		Anim.tween(TabIndicator, 0.32, { Position = indicatorPos }, Enum.EasingStyle.Quint)
	end

	-- page slides in from the side it came from (only on normal tab clicks, never in info-only mode)
	if not instant and not infoOnly and prevPage ~= pageIndex then
		local pages = { ParticlePage, AutoPage, FavPage, SavedPage }
		local page = pages[pageIndex]
		if page then
			local dir = (pageIndex > prevPage) and 1 or -1
			page.Position = UDim2.fromOffset(22 * dir, pageTopValue)
			PageTweens[page] = Anim.tween(page, 0.32, { Position = UDim2.fromOffset(0, pageTopValue) }, Enum.EasingStyle.Quint)
		end
	end

	AnimateRender = not instant
	if pageIndex == 2 then
		renderAutoPage()
	end
	if pageIndex == 3 then

-- layoutParticlePage is the ONE function that decides where everything sits.
-- It always sets every position and size from scratch, so it can be called at any time
-- (resize, info-only on/off, un-minimize) and the result is always the same clean layout.

layoutParticlePage = function(fullRebuild)
	if minimized then return end

	local pageW = math.max(currentW, MIN_W)
	local pageH = math.max(currentH, MIN_H)

	-- Single scale factor from a base size — everything shrinks/grows together
	local refW, refH = 460, 340
	if infoOnly then
		refW, refH = InfoMode.refW, InfoMode.refH
	end
	local scale = math.clamp(math.min(pageW / refW, pageH / refH), 0.65, 1.4)
	uiScale = scale
	RowHeight = math.max(26, math.floor(40 * scale + 0.5))

	local topH = math.max(28, math.floor(40 * scale + 0.5))
	local pad = math.max(4, math.floor(8 * scale + 0.5))
	local searchH = math.max(18, math.floor(24 * scale + 0.5))
	local tabH = math.max(20, math.floor(32 * scale + 0.5))
	local gap = math.max(4, math.floor(8 * scale + 0.5))

	local bw2 = math.max(32, math.floor(52 * scale + 0.5))

	local infoW = pageW - pad * 2 - gap - listW

	pageTopValue = pageTop
	ListViewH = contentH

	Min.TextSize = math.max(11, math.floor(14 * scale + 0.5))

		PageTweens[page] = nil

	PreviewButton.Position = UDim2.new(1, -(prevW + headerGap), 0, math.floor(4 * scale))
	FavButton.Position = UDim2.new(1, -(prevW + headerGap + favW + headerGap), 0, math.floor(4 * scale))

	local headerH = math.max(22, math.floor(28 * scale))

local resizing = false

applyMainSize = function(w, h, fullRebuild)
	if MainSizeTween then
		MainSizeTween:Cancel()
		MainSizeTween = nil
	end
	w = math.clamp(math.floor(w), MIN_W, MAX_W)
	h = math.clamp(math.floor(h), MIN_H, MAX_H)
	currentW = w
	currentH = h
	Main.Size = UDim2.fromOffset(w, h)
	if minimized then return end
	layoutParticlePage(fullRebuild == true)
end

-- Smoothly animates the window from a previous size to whatever it is now
local function tweenMainSizeFrom(fromW, fromH)
	local target = Main.Size
	Main.Size = UDim2.fromOffset(fromW, fromH)
	MainSizeTween = Anim.tween(Main, 0.32, { Size = target }, Enum.EasingStyle.Quint)
end

local function beginResize(input)
	if minimized then return end
	if MainSizeTween then
		MainSizeTween:Cancel()
		MainSizeTween = nil
		Main.Size = UDim2.fromOffset(currentW, currentH)
	end
	resizing = true
	resizeStart = input.Position
	resizeStartSize = Vector2.new(currentW, currentH)
end

local function endResize()
	resizing = false
	resizeStart = nil
	resizeStartSize = nil
	if not minimized then
		-- One full rebuild when user releases — textbox/buttons/text already scaled live
		applyMainSize(currentW, currentH, true)
	end
end

ResizeHandle.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		beginResize(input)
	end
end)

ResizeHandle.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		endResize()
	end
end)

-- ===================== INFO-ONLY MODE (second minimizer) =====================
-- Hides the whole browser (tabs, search, list, other pages) and shows only the info panel.

local function refreshInfoBtn()
	local gloss = InfoBtn:FindFirstChild("MetalGloss")
	if infoOnly then
		Anim.tween(InfoBtn, 0.2, {
			BackgroundTransparency = 0,
			BackgroundColor3 = C.accent,
			TextColor3 = Color3.new(1, 1, 1),
		})
		if gloss then gloss.Visible = true end
	else
		Anim.tween(InfoBtn, 0.2, {
			BackgroundTransparency = 1,
			TextColor3 = C.textMuted,
		})
		if gloss then gloss.Visible = false end
	end
end

local function setInfoOnly(on)
	if minimized then return end
	if on == infoOnly then return end

	local oldW, oldH = currentW, currentH
	local pageToShow = 1

	if on then
		InfoMode.normalW, InfoMode.normalH = currentW, currentH
		InfoMode.prevPage = currentPage
		infoOnly = true
		MIN_W, MIN_H = InfoMode.minW, InfoMode.minH
		currentW, currentH = InfoMode.w, InfoMode.h
		pageToShow = 1
	else
		InfoMode.w, InfoMode.h = currentW, currentH
		infoOnly = false
		MIN_W, MIN_H = InfoMode.normalMinW, InfoMode.normalMinH
		currentW, currentH = InfoMode.normalW, InfoMode.normalH
		pageToShow = InfoMode.prevPage or 1
	end

	refreshInfoBtn()
	-- FIX: do the layout FIRST (this sets the correct page position/size for the new mode),
	-- THEN switch the page instantly. Before, the page was switched first using the OLD
	-- position, so after leaving info-only the other tabs came back shifted and broken.
	applyMainSize(currentW, currentH, true)
	setPage(pageToShow, true)
	-- window glides to its new size
	tweenMainSizeFrom(oldW, oldH)
end

-- ===================== DRAGGING =====================

local dragging = false
local dragStart, startPosition

Top.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPosition = Main.Position
	end
end)

Top.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)

UIS.InputChanged:Connect(function(input)
	if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
		return
	end
	if resizing and resizeStart and resizeStartSize then
		local delta = input.Position - resizeStart
		local nw = resizeStartSize.X + delta.X
		local nh = resizeStartSize.Y + delta.Y
		local now = os.clock()
		if now - lastLiveLayout >= 0.04 then
			lastLiveLayout = now
			-- Live scale (no row recreate / no info rebuild) — keeps resize smooth
			applyMainSize(nw, nh, false)
		else
			nw = math.clamp(math.floor(nw), MIN_W, MAX_W)
			nh = math.clamp(math.floor(nh), MIN_H, MAX_H)
			currentW = nw
			currentH = nh
			Main.Size = UDim2.fromOffset(nw, nh)
		end
		return
	end
	if dragging and dragStart and startPosition then
		local delta = input.Position - dragStart
		Main.Position = UDim2.new(
			startPosition.X.Scale,
			startPosition.X.Offset + delta.X,
			startPosition.Y.Scale,
			startPosition.Y.Offset + delta.Y
		)
	end
end)

-- Safety net: if the mouse/finger is released away from the title bar or resize handle,
-- make sure dragging and resizing both stop.
UIS.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
		if resizing then
			endResize()
		end
	end
end)

-- ===================== AMBIENT ANIMATIONS =====================
-- Glowing rings that pulse while Auto Scan is on, a preview is playing, or a save is running,
-- plus a softly breathing title icon.

do
	local pulses = {}

	local function makePulse(btn, color, isActive)
		local s = Instance.new("UIStroke")
		s.Name = "PulseStroke"
		s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
		s.Color = color
		s.Thickness = 2
		s.Transparency = 1
		s.Parent = btn
		pulses[#pulses + 1] = { stroke = s, active = isActive }
	end

	makePulse(AutoToggleBtn, C.green, function() return AutoScanning end)
	makePulse(PreviewButton, C.red, function() return Previewing end)
	makePulse(Save, C.orange, function() return saving end)

	local startT = os.clock()
	local conn
	conn = RunService.Heartbeat:Connect(function()
		if not GUI.Parent then
			if conn then conn:Disconnect() end
			return
		end
		local t = os.clock() - startT
		local wave = 0.5 + 0.5 * math.sin(t * 5)
		for _, p in ipairs(pulses) do
			if p.stroke.Parent then
				if p.active() then
					p.stroke.Transparency = 0.25 + 0.5 * wave
				elseif p.stroke.Transparency ~= 1 then
					p.stroke.Transparency = 1
				end
			end
		end
		TitleIcon.TextColor3 = C.textMuted:Lerp(C.accent, 0.5 + 0.5 * math.sin(t * 1.6))
	end)
end
